1. ACTIVITY LIFECYCLE
What is Activity Lifecycle?
Android activities go through different states from creation to destruction. Understanding this helps you manage app behavior properly.

The 6 Main Lifecycle Methods:
Method	When it's called	What to do here
onCreate()	Activity first created	Initialize UI, setContentView, find views
onStart()	Activity becoming visible	Prepare to appear on screen
onResume()	Activity ready to interact	Start animations, acquire resources
onPause()	Another activity coming to front	Save data, release resources
onStop()	Activity no longer visible	Stop heavy operations
onDestroy()	Activity being destroyed	Clean up all resources

MainActivity.java:

java
package com.example.lifecycledemo;

import android.content.Intent;
import android.os.Bundle;
import android.util.Log;
import android.widget.Button;
import android.widget.TextView;
import android.widget.Toast;
import androidx.appcompat.app.AppCompatActivity;

public class MainActivity extends AppCompatActivity {
    
    private static final String TAG = "LifecycleDemo";
    private TextView textViewStatus;
    private Button btnNext;
    private StringBuilder statusLog = new StringBuilder();
    
    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);
        
        // Initialize views
        textViewStatus = findViewById(R.id.textViewStatus);
        btnNext = findViewById(R.id.btnNext);
        
        // Log and display status
        updateStatus("onCreate() called");
        Log.d(TAG, "onCreate called");
        Toast.makeText(this, "onCreate", Toast.LENGTH_SHORT).show();
        
        // Setup button to go to next activity
        btnNext.setOnClickListener(view -> {
            Intent intent = new Intent(MainActivity.this, SecondActivity.class);
            startActivity(intent);
        });
        
        // Check if there's saved instance state (like after rotation)
        if (savedInstanceState != null) {
            String savedText = savedInstanceState.getString("saved_text");
            updateStatus("Restored: " + savedText);
        }
    }
    
    @Override
    protected void onStart() {
        super.onStart();
        updateStatus("onStart() called");
        Log.d(TAG, "onStart called");
        Toast.makeText(this, "onStart", Toast.LENGTH_SHORT).show();
    }
    
    @Override
    protected void onResume() {
        super.onResume();
        updateStatus("onResume() called - Activity is now running");
        Log.d(TAG, "onResume called");
        Toast.makeText(this, "onResume", Toast.LENGTH_SHORT).show();
    }
    
    @Override
    protected void onPause() {
        super.onPause();
        updateStatus("onPause() called");
        Log.d(TAG, "onPause called");
        Toast.makeText(this, "onPause", Toast.LENGTH_SHORT).show();
    }
    
    @Override
    protected void onStop() {
        super.onStop();
        updateStatus("onStop() called");
        Log.d(TAG, "onStop called");
        Toast.makeText(this, "onStop", Toast.LENGTH_SHORT).show();
    }
    
    @Override
    protected void onRestart() {
        super.onRestart();
        updateStatus("onRestart() called");
        Log.d(TAG, "onRestart called");
        Toast.makeText(this, "onRestart", Toast.LENGTH_SHORT).show();
    }
    
    @Override
    protected void onDestroy() {
        super.onDestroy();
        updateStatus("onDestroy() called");
        Log.d(TAG, "onDestroy called");
        Toast.makeText(this, "onDestroy", Toast.LENGTH_SHORT).show();
    }
    
    // Save state before rotation
    @Override
    protected void onSaveInstanceState(Bundle outState) {
        super.onSaveInstanceState(outState);
        outState.putString("saved_text", statusLog.toString());
    }
    
    private void updateStatus(String event) {
        statusLog.append(event).append("\n");
        textViewStatus.setText(statusLog.toString());
    }
}

🔑 Key Points for Lifecycle:
onCreate() → onStart() → onResume() = Activity becomes visible and interactive

When you press Back: onPause() → onStop() → onDestroy()

2. INTENTS - NAVIGATION BETWEEN ACTIVITIES
Types of Intents:
A. Explicit Intent (Navigate within your app)
java
// Simple navigation
Intent intent = new Intent(CurrentActivity.this, TargetActivity.class);
startActivity(intent);

// Navigation with data
Intent intent = new Intent(MainActivity.this, SecondActivity.class);
intent.putExtra("KEY_NAME", "John Doe");
intent.putExtra("KEY_AGE", 25);
intent.putExtra("KEY_IS_STUDENT", true);
startActivity(intent);

// For result (when you want data back)
Intent intent = new Intent(MainActivity.this, SecondActivity.class);
startActivityForResult(intent, REQUEST_CODE); // REQUEST_CODE is any integer
B. Implicit Intent (Use other apps)
java
// Open a website
Intent intent = new Intent(Intent.ACTION_VIEW, 
                          Uri.parse("https://www.google.com"));
startActivity(intent);

// Make a phone call
Intent intent = new Intent(Intent.ACTION_DIAL, 
                          Uri.parse("tel:1234567890"));
startActivity(intent);

// Send an email
Intent intent = new Intent(Intent.ACTION_SENDTO, 
                          Uri.parse("mailto:example@email.com"));
intent.putExtra(Intent.EXTRA_SUBJECT, "Subject here");
intent.putExtra(Intent.EXTRA_TEXT, "Email body here");
startActivity(intent);

// Share text
Intent intent = new Intent(Intent.ACTION_SEND);
intent.setType("text/plain");
intent.putExtra(Intent.EXTRA_TEXT, "Shared text");
startActivity(Intent.createChooser(intent, "Share via"));

FirstActivity.java:

// Define request code
private static final int REQUEST_CODE = 100;

// Start activity for result
Intent intent = new Intent(MainActivity.this, SecondActivity.class);
startActivityForResult(intent, REQUEST_CODE);

// Handle the result
@Override
protected void onActivityResult(int requestCode, int resultCode, Intent data) {
    super.onActivityResult(requestCode, resultCode, data);
    
    if (requestCode == REQUEST_CODE) {
        if (resultCode == RESULT_OK) {
            String result = data.getStringExtra("RESULT_KEY");
            Toast.makeText(this, "Result: " + result, Toast.LENGTH_SHORT).show();
        } else if (resultCode == RESULT_CANCELED) {
            Toast.makeText(this, "Cancelled", Toast.LENGTH_SHORT).show();
        }
    }
}

    android:gravity="center_horizontal">

    <!-- orientation="vertical" stacks top to bottom -->
    <TextView
        android:layout_width="match_parent"
        android:layout_height="wrap_content"

        android:textStyle="bold"

    <EditText
        android:layout_width="match_parent"

        android:layout_marginTop="10dp"/>

    <EditText
        android:layout_width="match_parent"
        android:layout_height="wrap_content"
        android:hint="Enter email"
        android:layout_marginTop="10dp"/>

    <Button
        android:layout_width="match_parent"
        android:layout_height="wrap_content"
        android:text="Submit"
        android:layout_marginTop="20dp"/>

    <!-- Horizontal LinearLayout inside vertical -->
    <LinearLayout
        android:layout_width="match_parent"
        android:layout_height="wrap_content"
        android:orientation="horizontal"
        android:layout_marginTop="20dp"
        android:gravity="center">

        <Button
            android:layout_width="0dp"
            android:layout_height="wrap_content"
            android:layout_weight="1"

            android:layout_width="0dp"
            android:layout_height="wrap_content"
            android:layout_weight="1"
            android:text="OK"
            android:layout_marginStart="5dp"/>
    </LinearLayout>

</LinearLayout>
Key LinearLayout Attributes:

android:orientation="vertical|horizontal" - Direction

android:layout_weight - Distribute space proportionally (used with 0dp width/height)

android:gravity - Align children inside layout

android:layout_gravity - Align this view inside parent

layout_alignTop/Bottom/Left/Right - Align edges with another view

Key ConstraintLayout Attributes:

layout_constraintTop_toBottomOf - Top of this view aligns with bottom of reference

layout_constraintLeft_toLeftOf - Left of this view aligns with left of reference

layout_constraintRight_toRightOf - Right aligns with right

layout_constraintBottom_toTopOf - Bottom aligns with top of reference

Use 0dp width/height when constraining both sides (match constraint)

package com.example.fragmentdemo;

import android.content.Context;
import android.os.Bundle;
import android.view.LayoutInflater;
import android.view.View;
import android.view.ViewGroup;
import android.widget.Button;
import android.widget.EditText;
import android.widget.TextView;
import android.widget.Toast;
import androidx.annotation.NonNull;
import androidx.annotation.Nullable;
import androidx.fragment.app.Fragment;

public class ExampleFragment extends Fragment {
    
    private TextView fragmentTitle;
    private EditText fragmentInput;
    private Button fragmentButton;
    private FragmentListener listener;
    
    // Interface to communicate with Activity
    public interface FragmentListener {
        void onFragmentButtonClicked(String input);
    }
    
    // Create new instance of fragment with arguments
    public static ExampleFragment newInstance(String title) {
        ExampleFragment fragment = new ExampleFragment();
        Bundle args = new Bundle();
        args.putString("title", title);
        fragment.setArguments(args);
        return fragment;
    }
    
    @Override
    public void onAttach(@NonNull Context context) {
        super.onAttach(context);
        // Ensure the activity implements the listener
        if (context instanceof FragmentListener) {
            listener = (FragmentListener) context;
        } else {
            throw new RuntimeException(context.toString() + " must implement FragmentListener");
        }
    }
    
    @Override
    public void onCreate(@Nullable Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        // Retrieve arguments
        if (getArguments() != null) {
            String title = getArguments().getString("title");
        }
    }
    
    @Nullable
    @Override
    public View onCreateView(@NonNull LayoutInflater inflater, 
                             @Nullable ViewGroup container,
                             @Nullable Bundle savedInstanceState) {
        // Inflate the layout for this fragment
        return inflater.inflate(R.layout.fragment_example, container, false);
    }
    
    @Override
    public void onViewCreated(@NonNull View view, 
                              @Nullable Bundle savedInstanceState) {
        super.onViewCreated(view, savedInstanceState);
        
        // Initialize views
        fragmentTitle = view.findViewById(R.id.fragmentTitle);
        fragmentInput = view.findViewById(R.id.fragmentInput);
        fragmentButton = view.findViewById(R.id.fragmentButton);
        
        // Set title from arguments
        if (getArguments() != null) {
            fragmentTitle.setText(getArguments().getString("title"));
        }
        
        // Setup button click
        fragmentButton.setOnClickListener(v -> {
            String input = fragmentInput.getText().toString();
            // Show toast in fragment
            Toast.makeText(getActivity(), "Fragment: " + input, 
                          Toast.LENGTH_SHORT).show();
            // Send data to activity
            listener.onFragmentButtonClicked(input);
        });
    }
    
    @Override
    public void onDetach() {
        super.onDetach();
        listener = null;
    }
    
    // Public method to update fragment from activity
    public void updateFragmentText(String text) {
        if (fragmentTitle != null) {
            fragmentTitle.setText(text);
        }
    }
}

java
package com.example.fragmentdemo;

import android.os.Bundle;
import android.view.View;
import android.widget.Button;
import android.widget.Toast;
import androidx.appcompat.app.AppCompatActivity;
import androidx.fragment.app.Fragment;
import androidx.fragment.app.FragmentManager;
import androidx.fragment.app.FragmentTransaction;

public class MainActivity extends AppCompatActivity 
                          implements ExampleFragment.FragmentListener {
    
    private Button btnAddFragment, btnRemoveFragment;
    private FragmentManager fragmentManager;
    
    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);
        
        btnAddFragment = findViewById(R.id.btnAddFragment);
        btnRemoveFragment = findViewById(R.id.btnRemoveFragment);
        
        fragmentManager = getSupportFragmentManager();
        
        btnAddFragment.setOnClickListener(v -> addFragment());
        btnRemoveFragment.setOnClickListener(v -> removeFragment());
    }
    
    private void addFragment() {
        // Create fragment with arguments
        ExampleFragment fragment = ExampleFragment.newInstance("Dynamic Fragment");
        
        // Begin transaction
        FragmentTransaction transaction = fragmentManager.beginTransaction();
        
        // Add fragment to container
        transaction.add(R.id.fragmentContainer, fragment, "EXAMPLE_FRAGMENT");
        
        // Add to backstack (optional - allows back button to undo)
        transaction.addToBackStack(null);
        
        // Commit
        transaction.commit();
    }
    
    private void removeFragment() {
        // Find fragment by tag
        Fragment fragment = fragmentManager.findFragmentByTag("EXAMPLE_FRAGMENT");
        
        if (fragment != null) {
            FragmentTransaction transaction = fragmentManager.beginTransaction();
            transaction.remove(fragment);
            transaction.commit();
        }
    }
    
    // Implement FragmentListener interface
    @Override
    public void onFragmentButtonClicked(String input) {
        Toast.makeText(this, "Activity received: " + input, 
                      Toast.LENGTH_SHORT).show();
        
        // Find fragment and call its method
        ExampleFragment fragment = (ExampleFragment) 
            fragmentManager.findFragmentByTag("EXAMPLE_FRAGMENT");
        
        if (fragment != null) {
            fragment.updateFragmentText("Updated from activity: " + input);
        }
    }
}

CustomAdapter.java:

java
package com.example.listviewdemo;

import android.content.Context;
import android.view.LayoutInflater;
import android.view.View;
import android.view.ViewGroup;
import android.widget.ArrayAdapter;
import android.widget.ImageView;
import android.widget.TextView;
import androidx.annotation.NonNull;
import androidx.annotation.Nullable;
import java.util.List;

public class SportAdapter extends ArrayAdapter<Sport> {
    
    private Context context;
    private List<Sport> sportList;
    
    public SportAdapter(@NonNull Context context, @NonNull List<Sport> sportList) {
        super(context, 0, sportList);
        this.context = context;
        this.sportList = sportList;
    }
    
    @NonNull
    @Override
    public View getView(int position, @Nullable View convertView, 
                        @NonNull ViewGroup parent) {
        
        // ViewHolder pattern for performance (reuses views)
        ViewHolder holder;
        
        if (convertView == null) {
            // Inflate the custom layout
            convertView = LayoutInflater.from(context)
                .inflate(R.layout.list_item_sport, parent, false);
            
            // Create ViewHolder and store views
            holder = new ViewHolder();
            holder.imageView = convertView.findViewById(R.id.imageViewSport);
            holder.textViewName = convertView.findViewById(R.id.textViewSportName);
            holder.textViewCategory = convertView.findViewById(R.id.textViewSportCategory);
            holder.textViewPlayers = convertView.findViewById(R.id.textViewPlayers);
            
            // Store holder with view
            convertView.setTag(holder);
        } else {
            // Reuse existing holder
            holder = (ViewHolder) convertView.getTag();
        }
        
        // Get current sport object
        Sport currentSport = sportList.get(position);
        
        // Set data to views
        holder.imageView.setImageResource(currentSport.getImageResource());
        holder.textViewName.setText(currentSport.getName());
        holder.textViewCategory.setText("Category: " + currentSport.getCategory());
        holder.textViewPlayers.setText(currentSport.getPlayers() + " players");
        
        return convertView;
    }
    
    // ViewHolder class for performance
    private static class ViewHolder {
        ImageView imageView;
        TextView textViewName;
        TextView textViewCategory;
        TextView textViewPlayers;
    }
}

package com.example.tablelayoutdemo;

import android.os.Bundle;
import android.view.View;
import android.widget.Button;
import android.widget.EditText;
import android.widget.TableLayout;
import android.widget.TableRow;
import android.widget.TextView;
import android.widget.Toast;
import androidx.appcompat.app.AppCompatActivity;
import java.util.ArrayList;
import java.util.List;

public class DynamicTableActivity extends AppCompatActivity {
    
    private TableLayout tableLayout;
    private EditText editTextName, editTextPrice, editTextQuantity;
    private Button buttonAdd, buttonClear;
    private List<Product> productList;
    
    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_dynamic_table);
        
        tableLayout = findViewById(R.id.tableLayout);
        editTextName = findViewById(R.id.editTextName);
        editTextPrice = findViewById(R.id.editTextPrice);
        editTextQuantity = findViewById(R.id.editTextQuantity);
        buttonAdd = findViewById(R.id.buttonAdd);
        buttonClear = findViewById(R.id.buttonClear);
        
        productList = new ArrayList<>();
        
        // Add header row (only once)
        addHeaderRow();
        
        buttonAdd.setOnClickListener(v -> addProduct());
        buttonClear.setOnClickListener(v -> clearAllRows());
    }
    
    private void addHeaderRow() {
        TableRow headerRow = new TableRow(this);
        headerRow.setBackgroundColor(getResources().getColor(android.R.color.holo_blue_dark));
        
        // Create header cells
        TextView tvName = createHeaderTextView("Product");
        TextView tvPrice = createHeaderTextView("Price ($)");
        TextView tvQuantity = createHeaderTextView("Qty");
        TextView tvTotal = createHeaderTextView("Total ($)");
        TextView tvAction = createHeaderTextView("Action");
        
        // Set layout parameters
        tvName.setLayoutParams(new TableRow.LayoutParams(0, 
                               TableRow.LayoutParams.WRAP_CONTENT, 2f));
        tvPrice.setLayoutParams(new TableRow.LayoutParams(0, 
                                TableRow.LayoutParams.WRAP_CONTENT, 1f));
        tvQuantity.setLayoutParams(new TableRow.LayoutParams(0, 
                                  TableRow.LayoutParams.WRAP_CONTENT, 1f));
        tvTotal.setLayoutParams(new TableRow.LayoutParams(0, 
                                TableRow.LayoutParams.WRAP_CONTENT, 1f));
        tvAction.setLayoutParams(new TableRow.LayoutParams(0, 
                                 TableRow.LayoutParams.WRAP_CONTENT, 1f));
        
        // Add to row
        headerRow.addView(tvName);
        headerRow.addView(tvPrice);
        headerRow.addView(tvQuantity);
        headerRow.addView(tvTotal);
        headerRow.addView(tvAction);
        
        tableLayout.addView(headerRow);
    }
    
    private TextView createHeaderTextView(String text) {
        TextView textView = new TextView(this);
        textView.setText(text);
        textView.setTextColor(getResources().getColor(android.R.color.white));
        textView.setTextSize(16);
        textView.setPadding(16, 16, 16, 16);
        textView.setTypeface(null, android.graphics.Typeface.BOLD);
        return textView;
    }
    
    private TextView createDataTextView(String text, boolean isRightAligned) {
        TextView textView = new TextView(this);
        textView.setText(text);
        textView.setTextSize(14);
        textView.setPadding(16, 12, 16, 12);
        if (isRightAligned) {
            textView.setGravity(android.view.Gravity.END);
        }
        return textView;
    }
    
    private void addProduct() {
        // Get input values
        String name = editTextName.getText().toString().trim();
        String priceStr = editTextPrice.getText().toString().trim();
        String quantityStr = editTextQuantity.getText().toString().trim();
        
        // Validate inputs
        if (name.isEmpty() || priceStr.isEmpty() || quantityStr.isEmpty()) {
            Toast.makeText(this, "Please fill all fields", Toast.LENGTH_SHORT).show();
            return;
        }
        
        double price = Double.parseDouble(priceStr);
        int quantity = Integer.parseInt(quantityStr);
        double total = price * quantity;
        
        // Add to list
        Product product = new Product(name, price, quantity);
        productList.add(product);
        
        // Create new row
        addDataRow(name, price, quantity, total, productList.size() - 1);
        
        // Clear inputs
        editTextName.setText("");
        editTextPrice.setText("");
        editTextQuantity.setText("");
        
        // Show total at bottom
        updateTotalRow();
    }
    
    private void addDataRow(String name, double price, int quantity, 
                            double total, int position) {
        TableRow dataRow = new TableRow(this);
        dataRow.setPadding(0, 4, 0, 4);
        
        // Alternate row colors
        if (position % 2 == 0) {
            dataRow.setBackgroundColor(getResources().getColor(android.R.color.white));
        } else {
            dataRow.setBackgroundColor(getResources().getColor(android.R.color.background_light));
        }
        
        // Create data cells
        TextView tvName = createDataTextView(name, false);
        TextView tvPrice = createDataTextView(String.format("$%.2f", price), true);
        TextView tvQuantity = createDataTextView(String.valueOf(quantity), true);
        TextView tvTotal = createDataTextView(String.format("$%.2f", total), true);
        
        // Create delete button
        Button btnDelete = new Button(this);
        btnDelete.setText("Delete");
        btnDelete.setTextSize(12);
        btnDelete.setPadding(8, 4, 8, 4);
        btnDelete.setTag(position); // Store position as tag
        
        btnDelete.setOnClickListener(v -> {
            int pos = (int) v.getTag();
            deleteRow(pos);
        });
        
        // Set layout weights
        tvName.setLayoutParams(new TableRow.LayoutParams(0, 
                               TableRow.LayoutParams.WRAP_CONTENT, 2f));
        tvPrice.setLayoutParams(new TableRow.LayoutParams(0, 
                                TableRow.LayoutParams.WRAP_CONTENT, 1f));
        tvQuantity.setLayoutParams(new TableRow.LayoutParams(0, 
                                  TableRow.LayoutParams.WRAP_CONTENT, 1f));
        tvTotal.setLayoutParams(new TableRow.LayoutParams(0, 
                                TableRow.LayoutParams.WRAP_CONTENT, 1f));
        btnDelete.setLayoutParams(new TableRow.LayoutParams(0, 
                                  TableRow.LayoutParams.WRAP_CONTENT, 1f));
        
        // Add to row
        dataRow.addView(tvName);
        dataRow.addView(tvPrice);
        dataRow.addView(tvQuantity);
        dataRow.addView(tvTotal);
        dataRow.addView(btnDelete);
        
        tableLayout.addView(dataRow, tableLayout.getChildCount()); // Add at end
    }
    
    private void deleteRow(int position) {
        // Remove from list
        productList.remove(position);
        
        // Remove all data rows (keep header)
        int childCount = tableLayout.getChildCount();
        for (int i = childCount - 1; i > 0; i--) {
            tableLayout.removeViewAt(i);
        }
        
        // Re-add all remaining products
        for (int i = 0; i < productList.size(); i++) {
            Product p = productList.get(i);
            addDataRow(p.getName(), p.getPrice(), p.getQuantity(), 
                      p.getPrice() * p.getQuantity(), i);
        }
        
        // Update total
        updateTotalRow();
    }
    
    private void clearAllRows() {
        // Clear list
        productList.clear();
        
        // Remove all rows except header
        int childCount = tableLayout.getChildCount();
        for (int i = childCount - 1; i > 0; i--) {
            tableLayout.removeViewAt(i);
        }
        
        // Remove total row if exists
        TextView totalFooter = findViewById(R.id.textViewTotal);
        if (totalFooter != null) {
            ((ViewGroup) totalFooter.getParent()).removeView(totalFooter);
        }
    }

        // Calculate total
        double grandTotal = 0;
        for (Product p : productList) {
            grandTotal += p.getPrice() * p.getQuantity();

package com.example.togglebuttondemo;

import android.os.Bundle;
import android.widget.Button;
import android.widget.ImageView;
import android.widget.TextView;
import android.widget.ToggleButton;
import android.widget.Toast;
import androidx.appcompat.app.AppCompatActivity;

public class MainActivity extends AppCompatActivity {
    
    private ToggleButton toggleButton;
    private ImageView imageView;
    private TextView textViewStatus;
    private Button btnChangeMode;
    
    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);
        
        toggleButton = findViewById(R.id.toggleButton);
        imageView = findViewById(R.id.imageViewStatus);
        textViewStatus = findViewById(R.id.textViewStatus);
        btnChangeMode = findViewById(R.id.btnChangeMode);
        
        // Handle toggle button changes
        toggleButton.setOnCheckedChangeListener((buttonView, isChecked) -> {
            if (isChecked) {
                // Toggle is ON
                imageView.setImageResource(android.R.drawable.presence_online);
                textViewStatus.setText("Current Mode: Wi-Fi ON");
                Toast.makeText(this, "Wi-Fi Turned ON", Toast.LENGTH_SHORT).show();
            } else {
                // Toggle is OFF
                imageView.setImageResource(android.R.drawable.presence_offline);
                textViewStatus.setText("Current Mode: Wi-Fi OFF");
                Toast.makeText(this, "Wi-Fi Turned OFF", Toast.LENGTH_SHORT).show();
            }
        });
        
        // Change mode button
        btnChangeMode.setOnClickListener(v -> {
            // Toggle the state programmatically
            toggleButton.setChecked(!toggleButton.isChecked());
        });
    }
}

package com.example.checkboxdemo;

import android.os.Bundle;
import android.widget.Button;
import android.widget.CheckBox;
import android.widget.TextView;
import android.widget.Toast;
import androidx.appcompat.app.AppCompatActivity;
import java.util.ArrayList;

public class MainActivity extends AppCompatActivity {
    
    private CheckBox chkReading, chkMusic, chkSports, chkTravel;
    private Button btnSubmit;
    private TextView textViewResult;
    
    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);
        
        chkReading = findViewById(R.id.checkBoxReading);
        chkMusic = findViewById(R.id.checkBoxMusic);
        chkSports = findViewById(R.id.checkBoxSports);
        chkTravel = findViewById(R.id.checkBoxTravel);
        btnSubmit = findViewById(R.id.btnSubmit);
        textViewResult = findViewById(R.id.textViewResult);
        
        btnSubmit.setOnClickListener(v -> {
            ArrayList<String> selectedItems = new ArrayList<>();
            StringBuilder result = new StringBuilder("Selected: ");
            
            if (chkReading.isChecked()) {
                selectedItems.add("Reading");
                result.append("Reading, ");
            }
            if (chkMusic.isChecked()) {
                selectedItems.add("Music");
                result.append("Music, ");
            }
            if (chkSports.isChecked()) {
                selectedItems.add("Sports");
                result.append("Sports, ");
            }
            if (chkTravel.isChecked()) {
                selectedItems.add("Travel");
                result.append("Travel, ");
            }
            
            if (selectedItems.isEmpty()) {
                textViewResult.setText("No items selected");
                Toast.makeText(this, "Please select at least one hobby", 
                              Toast.LENGTH_SHORT).show();
            } else {
                // Remove last comma and space
                String finalResult = result.toString();
                finalResult = finalResult.substring(0, finalResult.length() - 2);
                textViewResult.setText(finalResult);
                
                // Show toast with count
                Toast.makeText(this, "Selected " + selectedItems.size() + 
                              " hobbies", Toast.LENGTH_SHORT).show();
            }
        });
    }
}

package com.example.radiodemo;

import android.os.Bundle;
import android.widget.Button;
import android.widget.RadioButton;
import android.widget.RadioGroup;
import android.widget.TextView;
import android.widget.Toast;
import androidx.appcompat.app.AppCompatActivity;

public class MainActivity extends AppCompatActivity {
    
    private RadioGroup radioGroupGender, radioGroupPayment;
    private RadioButton radioMale, radioFemale;
    private RadioButton radioCredit, radioDebit, radioPaypal, radioCash;
    private Button btnSubmit;
    private TextView textViewResult;
    
    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);
        
        radioGroupGender = findViewById(R.id.radioGroupGender);
        radioGroupPayment = findViewById(R.id.radioGroupPayment);
        radioMale = findViewById(R.id.radioMale);
        radioFemale = findViewById(R.id.radioFemale);
        radioCredit = findViewById(R.id.radioCredit);
        radioDebit = findViewById(R.id.radioDebit);
        radioPaypal = findViewById(R.id.radioPaypal);
        radioCash = findViewById(R.id.radioCash);
        btnSubmit = findViewById(R.id.btnSubmit);
        textViewResult = findViewById(R.id.textViewResult);
        
        // Listen for RadioGroup changes
        radioGroupGender.setOnCheckedChangeListener((group, checkedId) -> {
            if (checkedId == R.id.radioMale) {
                Toast.makeText(this, "Gender: Male selected", Toast.LENGTH_SHORT).show();
            } else if (checkedId == R.id.radioFemale) {
                Toast.makeText(this, "Gender: Female selected", Toast.LENGTH_SHORT).show();
            }
        });
        
        btnSubmit.setOnClickListener(v -> {
            String gender = "";
            String payment = "";
            
            // Get selected gender
            int selectedGenderId = radioGroupGender.getCheckedRadioButtonId();
            if (selectedGenderId == R.id.radioMale) {
                gender = "Male";
            } else if (selectedGenderId == R.id.radioFemale) {
                gender = "Female";
            } else {
                Toast.makeText(this, "Please select gender", Toast.LENGTH_SHORT).show();
                return;
            }
            
            // Get selected payment
            int selectedPaymentId = radioGroupPayment.getCheckedRadioButtonId();
            if (selectedPaymentId == R.id.radioCredit) {
                payment = "Credit Card";
            } else if (selectedPaymentId == R.id.radioDebit) {
                payment = "Debit Card";
            } else if (selectedPaymentId == R.id.radioPaypal) {
                payment = "PayPal";
            } else if (selectedPaymentId == R.id.radioCash) {
                payment = "Cash on Delivery";
            } else {
                Toast.makeText(this, "Please select payment method", 
                              Toast.LENGTH_SHORT).show();
                return;
            }
            
            // Display result
            String result = "Gender: " + gender + "\nPayment: " + payment;
            textViewResult.setText(result);
            
            Toast.makeText(this, "Submission successful!", Toast.LENGTH_SHORT).show();
        });
    }
}

package com.example.switchdemo;

import android.os.Bundle;
import android.widget.CompoundButton;
import android.widget.Switch;
import android.widget.TextView;
import androidx.appcompat.app.AppCompatActivity;

public class MainActivity extends AppCompatActivity {
    
    private Switch switchWifi, switchBluetooth, switchDarkMode;
    private TextView textViewStatus;
    
    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);
        
        switchWifi = findViewById(R.id.switchWifi);
        switchBluetooth = findViewById(R.id.switchBluetooth);
        switchDarkMode = findViewById(R.id.switchDarkMode);
        textViewStatus = findViewById(R.id.textViewStatus);
        
        // Individual listeners
        switchWifi.setOnCheckedChangeListener((buttonView, isChecked) -> {
            updateStatus();
            showToast("Wi-Fi " + (isChecked ? "ON" : "OFF"));
        });
        
        switchBluetooth.setOnCheckedChangeListener((buttonView, isChecked) -> {
            updateStatus();
            showToast("Bluetooth " + (isChecked ? "ON" : "OFF"));
        });
        
        switchDarkMode.setOnCheckedChangeListener((buttonView, isChecked) -> {
            updateStatus();
            showToast("Dark Mode " + (isChecked ? "ON" : "OFF"));
            
            // Optional: Change theme
            if (isChecked) {
                // Apply dark theme
                textViewStatus.setBackgroundColor(getResources().getColor(android.R.color.darker_gray));
                textViewStatus.setTextColor(getResources().getColor(android.R.color.white));
            } else {
                // Apply light theme
                textViewStatus.setBackgroundColor(getResources().getColor(android.R.color.background_light));
                textViewStatus.setTextColor(getResources().getColor(android.R.color.black));
            }
        });
    }
    
    private void updateStatus() {
        String status = "Status:\n" +
                       "Wi-Fi: " + (switchWifi.isChecked() ? "ON" : "OFF") + "\n" +
                       "Bluetooth: " + (switchBluetooth.isChecked() ? "ON" : "OFF") + "\n" +
                       "Dark Mode: " + (switchDarkMode.isChecked() ? "ON" : "OFF");
        
        textViewStatus.setText(status);
    }
    
    private void showToast(String message) {
        Toast.makeText(this, message, Toast.LENGTH_SHORT).show();
    }
}

package com.example.seekbardemo;

import android.os.Bundle;
import android.widget.SeekBar;
import android.widget.TextView;
import androidx.appcompat.app.AppCompatActivity;

public class MainActivity extends AppCompatActivity {
    
    private SeekBar seekBarVolume, seekBarBrightness;
    private TextView textViewVolume, textViewBrightness;
    
    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);
        
        seekBarVolume = findViewById(R.id.seekBarVolume);
        seekBarBrightness = findViewById(R.id.seekBarBrightness);
        textViewVolume = findViewById(R.id.textViewVolume);
        textViewBrightness = findViewById(R.id.textViewBrightness);
        
        // Volume SeekBar
        seekBarVolume.setOnSeekBarChangeListener(new SeekBar.OnSeekBarChangeListener() {
            @Override
            public void onProgressChanged(SeekBar seekBar, int progress, boolean fromUser) {
                textViewVolume.setText("Volume: " + progress + "%");
            }
            
            @Override
            public void onStartTrackingTouch(SeekBar seekBar) {
                // Called when user starts touching
                Toast.makeText(MainActivity.this, "Adjusting volume...", 
                              Toast.LENGTH_SHORT).show();
            }
            
            @Override
            public void onStopTrackingTouch(SeekBar seekBar) {
                // Called when user stops touching
                Toast.makeText(MainActivity.this, 
                              "Volume set to " + seekBar.getProgress() + "%", 
                              Toast.LENGTH_SHORT).show();
            }
        });
        
        // Brightness SeekBar
        seekBarBrightness.setOnSeekBarChangeListener(new SeekBar.OnSeekBarChangeListener() {
            @Override
            public void onProgressChanged(SeekBar seekBar, int progress, boolean fromUser) {
                textViewBrightness.setText("Brightness: " + progress + "%");
            }
            
            @Override
            public void onStartTrackingTouch(SeekBar seekBar) {
                Toast.makeText(MainActivity.this, "Adjusting brightness...", 
                              Toast.LENGTH_SHORT).show();
            }
            
            @Override
            public void onStopTrackingTouch(SeekBar seekBar) {
                Toast.makeText(MainActivity.this, 
                              "Brightness set to " + seekBar.getProgress() + "%", 
                              Toast.LENGTH_SHORT).show();
            }
        });
    }
}

package com.example.testapp;

import android.os.Bundle;
import android.view.Gravity;
import android.view.LayoutInflater;
import android.view.View;
import android.widget.Button;
import android.widget.ImageView;
import android.widget.TextView;
import android.widget.Toast;
import android.widget.ToggleButton;
import androidx.appcompat.app.AppCompatActivity;

public class MainActivity extends AppCompatActivity {
    
    private Button btnShowToast;
    private ToggleButton toggleButton;
    
    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);
        
        btnShowToast = findViewById(R.id.btnShowToast);
        toggleButton = findViewById(R.id.toggleButton);
        
        btnShowToast.setOnClickListener(v -> showCustomToast("Button Clicked!", 
                                     android.R.drawable.ic_dialog_info));
        
        toggleButton.setOnCheckedChangeListener((buttonView, isChecked) -> {
            if (isChecked) {
                showCustomToast("Toggle ON", android.R.drawable.presence_online);
            } else {
                showCustomToast("Toggle OFF", android.R.drawable.presence_offline);
            }
        });
    }
    
    private void showCustomToast(String message, int iconRes) {
        // Inflate custom layout
        LayoutInflater inflater = getLayoutInflater();
        View layout = inflater.inflate(R.layout.custom_toast, 
                                       findViewById(R.id.toast_layout_root));
        
        // Set image and text
        ImageView imageView = layout.findViewById(R.id.imageViewToast);
        TextView textView = layout.findViewById(R.id.textViewToast);
        
        imageView.setImageResource(iconRes);
        textView.setText(message);
        
        // Create and show toast
        Toast toast = new Toast(getApplicationContext());
        toast.setDuration(Toast.LENGTH_SHORT);
        toast.setView(layout);
        toast.setGravity(Gravity.CENTER, 0, 0);
        toast.show();
    }
}

package com.example.spinnerdemo;

import android.os.Bundle;
import android.view.View;
import android.widget.AdapterView;
import android.widget.ArrayAdapter;
import android.widget.Button;
import android.widget.Spinner;
import android.widget.TextView;
import android.widget.Toast;
import androidx.appcompat.app.AppCompatActivity;

public class MainActivity extends AppCompatActivity {
    
    private Spinner spinnerVehicle;
    private Button btnSubmit;
    private TextView textViewResult;
    private String selectedVehicle;
    
    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);
        
        spinnerVehicle = findViewById(R.id.spinnerVehicle);
        btnSubmit = findViewById(R.id.btnSubmit);
        textViewResult = findViewById(R.id.textViewResult);
        
        // Data for spinner
        String[] vehicles = {"Select Vehicle Type", "Car", "Bike", "Truck", "Bus", "Bicycle"};
        
        // Create ArrayAdapter
        ArrayAdapter<String> adapter = new ArrayAdapter<>(
            this,
            android.R.layout.simple_spinner_item,  // Layout for spinner closed state
            vehicles
        );
        
        // Set dropdown layout
        adapter.setDropDownViewResource(android.R.layout.simple_spinner_dropdown_item);
        
        // Set adapter to spinner
        spinnerVehicle.setAdapter(adapter);
        
        // Set default selection (optional)
        spinnerVehicle.setSelection(0);
        
        // Handle spinner item selection
        spinnerVehicle.setOnItemSelectedListener(new AdapterView.OnItemSelectedListener() {
            @Override
            public void onItemSelected(AdapterView<?> parent, View view, int position, long id) {
                selectedVehicle = parent.getItemAtPosition(position).toString();
                
                if (position == 0) {
                    // First item (prompt) - don't show toast
                    selectedVehicle = null;
                } else {
                    Toast.makeText(MainActivity.this, 
                                  "Selected: " + selectedVehicle, 
                                  Toast.LENGTH_SHORT).show();
                }
            }
            
            @Override
            public void onNothingSelected(AdapterView<?> parent) {
                selectedVehicle = null;
            }
        });
        
        // Submit button click
        btnSubmit.setOnClickListener(v -> {
            if (selectedVehicle == null || selectedVehicle.equals("Select Vehicle Type")) {
                Toast.makeText(MainActivity.this, 
                              "Please select a vehicle type", 
                              Toast.LENGTH_SHORT).show();
            } else {
                textViewResult.setText("Selected Vehicle: " + selectedVehicle);
            }
        });
    }
}

Custom Spinner Adapter (VehicleAdapter.java):

java
package com.example.spinnerdemo;

import android.content.Context;
import android.view.LayoutInflater;
import android.view.View;
import android.view.ViewGroup;
import android.widget.ArrayAdapter;
import android.widget.ImageView;
import android.widget.TextView;
import androidx.annotation.NonNull;
import androidx.annotation.Nullable;
import java.util.List;

public class VehicleAdapter extends ArrayAdapter<Vehicle> {
    
    private Context context;
    private List<Vehicle> vehicleList;
    
    public VehicleAdapter(@NonNull Context context, @NonNull List<Vehicle> vehicleList) {
        super(context, 0, vehicleList);
        this.context = context;
        this.vehicleList = vehicleList;
    }
    
    @NonNull
    @Override
    public View getView(int position, @Nullable View convertView, @NonNull ViewGroup parent) {
        // This is for the selected item view
        return createView(position, convertView, parent);
    }
    
    @Override
    public View getDropDownView(int position, @Nullable View convertView, @NonNull ViewGroup parent) {
        // This is for the dropdown items
        return createView(position, convertView, parent);
    }
    
    private View createView(int position, View convertView, ViewGroup parent) {
        ViewHolder holder;
        
        if (convertView == null) {
            convertView = LayoutInflater.from(context)
                .inflate(R.layout.spinner_item_custom, parent, false);
            
            holder = new ViewHolder();
            holder.imageView = convertView.findViewById(R.id.imageViewVehicle);
            holder.textViewName = convertView.findViewById(R.id.textViewName);
            holder.textViewType = convertView.findViewById(R.id.textViewType);
            
            convertView.setTag(holder);
        } else {
            holder = (ViewHolder) convertView.getTag();
        }
        
        Vehicle vehicle = vehicleList.get(position);
        
        holder.imageView.setImageResource(vehicle.getImageResource());
        holder.textViewName.setText(vehicle.getName());
        holder.textViewType.setText(vehicle.getType());
        
        return convertView;
    }
    
    private static class ViewHolder {
        ImageView imageView;
        TextView textViewName;
        TextView textViewType;
    }
}

package com.example.pickerdemo;

import android.app.DatePickerDialog;
import android.app.TimePickerDialog;
import android.os.Bundle;
import android.view.View;
import android.widget.Button;
import android.widget.TextView;
import android.widget.Toast;
import androidx.appcompat.app.AppCompatActivity;
import java.text.SimpleDateFormat;
import java.util.Calendar;
import java.util.Locale;

public class MainActivity extends AppCompatActivity {
    
    private Button btnSelectDate, btnSelectTime;
    private TextView textViewDate, textViewTime;
    private Calendar calendar;
    private SimpleDateFormat dateFormat, timeFormat;
    
    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);
        
        btnSelectDate = findViewById(R.id.btnSelectDate);
        btnSelectTime = findViewById(R.id.btnSelectTime);
        textViewDate = findViewById(R.id.textViewDate);
        textViewTime = findViewById(R.id.textViewTime);
        
        calendar = Calendar.getInstance();
        dateFormat = new SimpleDateFormat("dd/MM/yyyy", Locale.getDefault());
        timeFormat = new SimpleDateFormat("hh:mm a", Locale.getDefault());
        
        // Display current date and time
        updateDateDisplay();
        updateTimeDisplay();
        
        btnSelectDate.setOnClickListener(v -> showDatePicker());
        btnSelectTime.setOnClickListener(v -> showTimePicker());
    }
    
    private void showDatePicker() {
        int year = calendar.get(Calendar.YEAR);
        int month = calendar.get(Calendar.MONTH);
        int day = calendar.get(Calendar.DAY_OF_MONTH);
        
        DatePickerDialog datePickerDialog = new DatePickerDialog(
            this,
            (view, selectedYear, selectedMonth, selectedDay) -> {
                // Set the selected date to calendar
                calendar.set(Calendar.YEAR, selectedYear);
                calendar.set(Calendar.MONTH, selectedMonth);
                calendar.set(Calendar.DAY_OF_MONTH, selectedDay);
                
                // Update display
                updateDateDisplay();
                
                Toast.makeText(MainActivity.this, 
                              "Date selected: " + dateFormat.format(calendar.getTime()), 
                              Toast.LENGTH_SHORT).show();
            },
            year, month, day
        );
        
        // Optional: Set min and max date
        // datePickerDialog.getDatePicker().setMinDate(System.currentTimeMillis() - 1000);
        
        datePickerDialog.show();
    }
    
    private void showTimePicker() {
        int hour = calendar.get(Calendar.HOUR_OF_DAY);
        int minute = calendar.get(Calendar.MINUTE);
        
        TimePickerDialog timePickerDialog = new TimePickerDialog(
            this,
            (view, hourOfDay, minuteOfHour) -> {
                // Set the selected time to calendar
                calendar.set(Calendar.HOUR_OF_DAY, hourOfDay);
                calendar.set(Calendar.MINUTE, minuteOfHour);
                
                // Update display
                updateTimeDisplay();
                
                Toast.makeText(MainActivity.this, 
                              "Time selected: " + timeFormat.format(calendar.getTime()), 
                              Toast.LENGTH_SHORT).show();
            },
            hour, minute, false  // false = 12-hour format, true = 24-hour format
        );
        
        timePickerDialog.show();
    }
    
    private void updateDateDisplay() {
        textViewDate.setText("Selected Date: " + dateFormat.format(calendar.getTime()));
    }
    
    private void updateTimeDisplay() {
        textViewTime.setText("Selected Time: " + timeFormat.format(calendar.getTime()));
    }
}

            }
        });

        vehicleNo = editTextVehicleNo.getText().toString().trim();
        rcNo = editTextRCNo.getText().toString().trim();
        
        if (selectedVehicleType == null) {

        }

            editTextVehicleNo.setError("Vehicle number required");

        if (rcNo.isEmpty()) {
            editTextRCNo.setError("RC number required");
            return false;
        }

    private void showConfirmationDialog() {
        // Build dialog
        AlertDialog.Builder builder = new AlertDialog.Builder(this);
        LayoutInflater inflater = getLayoutInflater();
        View dialogView = inflater.inflate(R.layout.dialog_confirm, null);
        builder.setView(dialogView);
        
        // Get dialog views
        TextView textViewType = dialogView.findViewById(R.id.textViewVehicleType);
        TextView textViewVehicleNo = dialogView.findViewById(R.id.textViewVehicleNo);

    }

        spinnerVehicleType.setSelection(0);
        editTextVehicleNo.setText("");
        editTextRCNo.setText("");
    }

    }

java
package com.example.toolbardemo;

import android.os.Bundle;
import android.view.Menu;
import android.view.MenuItem;
import android.widget.TextView;
import android.widget.Toast;
import androidx.appcompat.app.AppCompatActivity;
import androidx.appcompat.widget.Toolbar;

public class MainActivity extends AppCompatActivity {
    
    private Toolbar toolbar;
    private TextView textViewDisplay;
    
    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);
        
        // Initialize views
        toolbar = findViewById(R.id.toolbar);
        textViewDisplay = findViewById(R.id.textViewDisplay);
        
        // Set toolbar as action bar
        setSupportActionBar(toolbar);
        
        // Enable back button (optional)
        if (getSupportActionBar() != null) {
            getSupportActionBar().setDisplayHomeAsUpEnabled(true);
            getSupportActionBar().setHomeButtonEnabled(true);
        }
        
        // Set toolbar navigation click listener
        toolbar.setNavigationOnClickListener(v -> {
            textViewDisplay.setText("Navigation icon clicked");
            Toast.makeText(this, "Back button clicked", Toast.LENGTH_SHORT).show();
        });
    }
    
    @Override
    public boolean onCreateOptionsMenu(Menu menu) {
        getMenuInflater().inflate(R.menu.menu_main, menu);
        return true;
    }
    
    @Override
    public boolean onOptionsItemSelected(MenuItem item) {
        int id = item.getItemId();
        
        if (id == android.R.id.home) {
            // Handle back button
            textViewDisplay.setText("Home/Back clicked");
            Toast.makeText(this, "Back button clicked", Toast.LENGTH_SHORT).show();
            return true;
        }
        
        // Handle other menu items
        switch (id) {
            case R.id.action_search:
                textViewDisplay.setText("Search clicked");
                Toast.makeText(this, "Search clicked", Toast.LENGTH_SHORT).show();
                return true;
                
            case R.id.action_settings:
                textViewDisplay.setText("Settings clicked");
                Toast.makeText(this, "Settings clicked", Toast.LENGTH_SHORT).show();
                return true;
                
            default:
                return super.onOptionsItemSelected(item);
        }
    }
}

package com.example.contextmenudemo;

import android.os.Bundle;
import android.view.ContextMenu;
import android.view.MenuInflater;
import android.view.MenuItem;
import android.view.View;
import android.widget.TextView;
import android.widget.Toast;
import androidx.appcompat.app.AppCompatActivity;

public class MainActivity extends AppCompatActivity {
    
    private TextView textView1, textView2, textView3, textViewResult;
    private View longPressedView;
    
    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);
        
        textView1 = findViewById(R.id.textView1);
        textView2 = findViewById(R.id.textView2);
        textView3 = findViewById(R.id.textView3);
        textViewResult = findViewById(R.id.textViewResult);
        
        // Register views for context menu
        registerForContextMenu(textView1);
        registerForContextMenu(textView2);
        registerForContextMenu(textView3);
    }
    
    // This method creates the context menu when view is long-pressed
    @Override
    public void onCreateContextMenu(ContextMenu menu, View v, 
                                    ContextMenu.ContextMenuInfo menuInfo) {
        super.onCreateContextMenu(menu, v, menuInfo);
        
        // Store which view was long pressed
        longPressedView = v;
        
        // Set menu title based on which view was pressed
        if (v.getId() == R.id.textView1) {
            menu.setHeaderTitle("Actions for Item 1");
            menu.setHeaderIcon(android.R.drawable.ic_dialog_info);
        } else if (v.getId() == R.id.textView2) {
            menu.setHeaderTitle("Actions for Item 2");
            menu.setHeaderIcon(android.R.drawable.ic_dialog_info);
        } else if (v.getId() == R.id.textView3) {
            menu.setHeaderTitle("Actions for Item 3");
            menu.setHeaderIcon(android.R.drawable.ic_dialog_info);
        }
        
        // Inflate the menu
        MenuInflater inflater = getMenuInflater();
        inflater.inflate(R.menu.context_menu, menu);
    }
    
    // This method handles context menu item clicks
    @Override
    public boolean onContextItemSelected(MenuItem item) {
        String itemText = "";
        
        // Get the text of the long-pressed view
        if (longPressedView instanceof TextView) {
            itemText = ((TextView) longPressedView).getText().toString();
        }
        
        // Handle menu item clicks
        int id = item.getItemId();
        
        if (id == R.id.action_edit) {
            String result = "Edit clicked on: " + itemText;
            textViewResult.setText(result);
            Toast.makeText(this, "Editing " + itemText, Toast.LENGTH_SHORT).show();
            return true;
            
        } else if (id == R.id.action_delete) {
            String result = "Delete clicked on: " + itemText;
            textViewResult.setText(result);
            Toast.makeText(this, "Deleting " + itemText, Toast.LENGTH_SHORT).show();
            
            // Optionally hide the view
            longPressedView.setVisibility(View.GONE);
            return true;
            
        } else if (id == R.id.action_share) {
            String result = "Share clicked on: " + itemText;
            textViewResult.setText(result);
            Toast.makeText(this, "Sharing " + itemText, Toast.LENGTH_SHORT).show();
            return true;
            
        } else if (id == R.id.action_details) {
            String result = "Details clicked on: " + itemText;
            textViewResult.setText(result);
            Toast.makeText(this, "Viewing details of " + itemText, Toast.LENGTH_SHORT).show();
            return true;
            
        } else {
            return super.onContextItemSelected(item);
        }
    }

java
package com.example.cabdemo;

import android.os.Bundle;
import android.view.ActionMode;
import android.view.Menu;
import android.view.MenuInflater;
import android.view.MenuItem;
import android.view.View;
import android.widget.AbsListView;
import android.widget.AdapterView;
import android.widget.ArrayAdapter;
import android.widget.ListView;
import android.widget.TextView;
import android.widget.Toast;
import androidx.appcompat.app.AppCompatActivity;
import java.util.ArrayList;

public class CABActivity extends AppCompatActivity {
    
    private ListView listView;
    private TextView textViewSelection;
    private ArrayAdapter<String> adapter;
    private ArrayList<String> itemList;
    private ArrayList<String> selectedItems = new ArrayList<>();
    private ActionMode actionMode;
    
    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_cab);
        
        listView = findViewById(R.id.listView);
        textViewSelection = findViewById(R.id.textViewSelection);
        
        // Prepare data
        itemList = new ArrayList<>();
        for (int i = 1; i <= 20; i++) {
            itemList.add("Item " + i);
        }
        
        // Create adapter
        adapter = new ArrayAdapter<>(
            this,
            android.R.layout.simple_list_item_activated_1,  // For selection highlighting
            itemList
        );
        listView.setAdapter(adapter);
        
        // Enable multiple choice mode
        listView.setChoiceMode(ListView.CHOICE_MODE_MULTIPLE_MODAL);
        
        // Set multi-choice mode listener
        listView.setMultiChoiceModeListener(new AbsListView.MultiChoiceModeListener() {
            
            @Override
            public void onItemCheckedStateChanged(ActionMode mode, int position, 
                                                  long id, boolean checked) {
                // Update selected items count
                int selectedCount = listView.getCheckedItemCount();
                
                // Update selected items list
                String item = itemList.get(position);
                if (checked) {
                    selectedItems.add(item);
                } else {
                    selectedItems.remove(item);
                }
                
                // Update title with count
                mode.setTitle(selectedCount + " item(s) selected");
                
                // Update subtitle with selected items
                updateSelectionDisplay();
            }
            
            @Override
            public boolean onCreateActionMode(ActionMode mode, Menu menu) {
                // Inflate menu for CAB
                MenuInflater inflater = mode.getMenuInflater();
                inflater.inflate(R.menu.cab_menu, menu);
                
                // Store action mode reference
                actionMode = mode;
                
                // Show selection display
                textViewSelection.setVisibility(View.VISIBLE);
                
                return true;
            }
            
            @Override
            public boolean onPrepareActionMode(ActionMode mode, Menu menu) {
                return false;
            }
            
            @Override
            public boolean onActionItemClicked(ActionMode mode, MenuItem item) {
                int id = item.getItemId();
                
                if (id == R.id.action_delete) {
                    deleteSelectedItems();
                    mode.finish(); // Close CAB
                    return true;
                    
                } else if (id == R.id.action_share) {
                    shareSelectedItems();
                    mode.finish();
                    return true;
                    
                } else if (id == R.id.action_select_all) {
                    selectAllItems();
                    return true;
                    
                } else if (id == R.id.action_clear) {
                    clearSelections();
                    return true;
                }
                
                return false;
            }
            
            @Override
            public void onDestroyActionMode(ActionMode mode) {
                // Clear selections
                selectedItems.clear();
                listView.clearChoices();
                adapter.notifyDataSetChanged();
                
                // Hide selection display
                textViewSelection.setVisibility(View.GONE);
                textViewSelection.setText("No items selected");
                
                actionMode = null;
            }
        });
        
        // Handle regular item clicks
        listView.setOnItemClickListener(new AdapterView.OnItemClickListener() {
            @Override
            public void onItemClick(AdapterView<?> parent, View view, 
                                    int position, long id) {
                Toast.makeText(CABActivity.this, 
                              "Clicked: " + itemList.get(position), 
                              Toast.LENGTH_SHORT).show();
            }
        });
    }
    
    private void deleteSelectedItems() {
        // Remove selected items from list
        for (String item : selectedItems) {
            itemList.remove(item);
        }
        
        // Update adapter
        adapter.notifyDataSetChanged();
        
        String message = "Deleted " + selectedItems.size() + " item(s)";
        Toast.makeText(this, message, Toast.LENGTH_SHORT).show();
    }
    
    private void shareSelectedItems() {
        StringBuilder items = new StringBuilder();
        for (String item : selectedItems) {
            items.append(item).append(", ");
        }
        
        // Remove last comma
        if (items.length() > 0) {
            items.setLength(items.length() - 2);
        }
        
        String message = "Sharing: " + items.toString();
        Toast.makeText(this, message, Toast.LENGTH_SHORT).show();
    }
    
    private void selectAllItems() {
        int count = listView.getCount();
        for (int i = 0; i < count; i++) {
            listView.setItemChecked(i, true);
        }
    }
    
    private void clearSelections() {
        listView.clearChoices();
        adapter.notifyDataSetChanged();
        selectedItems.clear();
        if (actionMode != null) {
            actionMode.setTitle("0 item(s) selected");
        }
    }
    
    private void updateSelectionDisplay() {
        StringBuilder display = new StringBuilder("Selected: ");
        for (String item : selectedItems) {
            display.append(item).append(", ");
        }
        
        if (selectedItems.isEmpty()) {
            display.append("None");
        } else {
            // Remove last comma
            display.setLength(display.length() - 2);
        }
        
        textViewSelection.setText(display.toString());
    }
}

package com.example.digitalapp;

import android.graphics.Color;
import android.os.Bundle;
import android.text.Spannable;
import android.text.SpannableString;
import android.text.style.BackgroundColorSpan;
import android.view.MenuItem;
import android.view.View;
import android.widget.Button;
import android.widget.EditText;
import android.widget.PopupMenu;
import android.widget.TextView;
import android.widget.Toast;
import androidx.appcompat.app.AppCompatActivity;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.Collections;
import java.util.Comparator;
import java.util.List;

public class DigitalTransformationActivity extends AppCompatActivity {
    
    private Button btnFilter, btnSearch;
    private EditText editTextSearch;
    private TextView textViewContent, textViewStatus;
    private String originalText;
    private String currentText;
    private List<String> searchKeywords = new ArrayList<>();
    private boolean isHighlighted = false;
    
    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_digital_transformation);
        
        btnFilter = findViewById(R.id.btnFilter);
        btnSearch = findViewById(R.id.btnSearch);
        editTextSearch = findViewById(R.id.editTextSearch);
        textViewContent = findViewById(R.id.textViewContent);
        textViewStatus = findViewById(R.id.textViewStatus);
        
        originalText = getString(R.string.digital_transformation_text);
        currentText = originalText;
        textViewContent.setText(originalText);
        
        btnFilter.setOnClickListener(new View.OnClickListener() {
            @Override
            public void onClick(View v) {
                showFilterMenu(v);
            }
        });
        
        btnSearch.setOnClickListener(new View.OnClickListener() {
            @Override
            public void onClick(View v) {
                performSearch();
            }
        });
    }
    
    private void showFilterMenu(View anchor) {
        PopupMenu popupMenu = new PopupMenu(this, anchor);
        popupMenu.getMenuInflater().inflate(R.menu.filter_menu, popupMenu.getMenu());
        
        popupMenu.setOnMenuItemClickListener(new PopupMenu.OnMenuItemClickListener() {
            @Override
            public boolean onMenuItemClick(MenuItem item) {
                int id = item.getItemId();
                
                if (id == R.id.action_search) {
                    showSearchBox();
                    return true;
                    
                } else if (id == R.id.action_highlight) {
                    toggleHighlight();
                    return true;
                    
                } else if (id == R.id.action_sort_alpha) {
                    sortAlphabetically();
                    return true;
                    
                } else if (id == R.id.action_sort_relevance) {
                    sortByRelevance();
                    return true;
                }
                
                return false;
            }
        });
        
        popupMenu.show();
    }
    
    private void showSearchBox() {
        editTextSearch.setVisibility(View.VISIBLE);
        btnSearch.setVisibility(View.VISIBLE);
        textViewStatus.setText("Enter keywords to search (comma-separated)");
    }
    
    private void performSearch() {
        String input = editTextSearch.getText().toString().trim();
        
        if (input.isEmpty()) {
            Toast.makeText(this, "Please enter search keywords", 
                          Toast.LENGTH_SHORT).show();
            return;
        }
        
        // Parse keywords
        String[] keywords = input.split(",");
        searchKeywords.clear();
        for (String keyword : keywords) {
            searchKeywords.add(keyword.trim().toLowerCase());
        }
        
        // Perform search
        StringBuilder result = new StringBuilder();
        int totalMatches = 0;
        
        for (String keyword : searchKeywords) {
            int count = countOccurrences(originalText.toLowerCase(), keyword);
            totalMatches += count;
            result.append(keyword).append(": ").append(count).append(" matches\n");
        }
        
        textViewStatus.setText("Search Results:\n" + result.toString());
        
        // Highlight if enabled
        if (isHighlighted) {
            highlightKeywords();
        }
        
        Toast.makeText(this, "Found " + totalMatches + " total matches", 
                      Toast.LENGTH_SHORT).show();
    }
    
    private int countOccurrences(String text, String keyword) {
        int count = 0;
        int index = 0;
        
        while ((index = text.indexOf(keyword, index)) != -1) {
            count++;
            index += keyword.length();
        }
        
        return count;
    }
    
    private void toggleHighlight() {
        isHighlighted = !isHighlighted;
        
        if (isHighlighted && !searchKeywords.isEmpty()) {
            highlightKeywords();
            textViewStatus.setText("Highlight mode: ON");
            Toast.makeText(this, "Highlight enabled", Toast.LENGTH_SHORT).show();
        } else {
            // Remove highlighting
            textViewContent.setText(originalText);
            currentText = originalText;
            textViewStatus.setText("Highlight mode: OFF");
            Toast.makeText(this, "Highlight disabled", Toast.LENGTH_SHORT).show();
        }
    }
    
    private void highlightKeywords() {
        SpannableString spannable = new SpannableString(originalText);
        
        for (String keyword : searchKeywords) {
            String text = originalText.toLowerCase();
            String lowerKeyword = keyword.toLowerCase();
            
            int index = 0;
            while ((index = text.indexOf(lowerKeyword, index)) != -1) {
                spannable.setSpan(
                    new BackgroundColorSpan(Color.YELLOW),
                    index,
                    index + keyword.length(),
                    Spannable.SPAN_EXCLUSIVE_EXCLUSIVE
                );
                index += keyword.length();
            }
        }
        
        textViewContent.setText(spannable);
        currentText = originalText;
    }
    
    private void sortAlphabetically() {
        // Split into sentences or words
        String[] sentences = originalText.split("\\. ");
        List<String> sentenceList = new ArrayList<>(Arrays.asList(sentences));
        
        // Sort alphabetically
        Collections.sort(sentenceList, new Comparator<String>() {
            @Override
            public int compare(String s1, String s2) {
                return s1.compareToIgnoreCase(s2);
            }
        });
        
        // Rebuild text
        StringBuilder sorted = new StringBuilder();
        for (String sentence : sentenceList) {
            sorted.append(sentence).append(". ");
        }
        
        // Remove last period
        if (sorted.length() > 0) {
            sorted.setLength(sorted.length() - 1);
        }
        
        textViewContent.setText(sorted.toString());
        currentText = sorted.toString();
        
        textViewStatus.setText("Sorted alphabetically");
        Toast.makeText(this, "Content sorted alphabetically", 
                      Toast.LENGTH_SHORT).show();
        
        // Re-apply highlight if needed
        if (isHighlighted && !searchKeywords.isEmpty()) {
            highlightKeywords();
        }
    }
    
    private void sortByRelevance() {
        if (searchKeywords.isEmpty()) {
            Toast.makeText(this, "Please search for keywords first", 
                          Toast.LENGTH_SHORT).show();
            return;
        }
        
        // Split into sentences
        String[] sentences = originalText.split("\\. ");
        List<RelevantSentence> relevantSentences = new ArrayList<>();
        
        // Calculate relevance score for each sentence
        for (String sentence : sentences) {
            int score = 0;
            String lowerSentence = sentence.toLowerCase();
            
            for (String keyword : searchKeywords) {
                int count = countOccurrences(lowerSentence, keyword);
                score += count * 10; // Each occurrence adds 10 points
            }
            
            // Add points for containing keywords in title case (approximation)
            for (String keyword : searchKeywords) {
                if (lowerSentence.contains(keyword)) {
                    score += 5; // Bonus for having the keyword
                }
            }
            
            relevantSentences.add(new RelevantSentence(sentence, score));
        }
        
        // Sort by relevance (higher score first)
        Collections.sort(relevantSentences, new Comparator<RelevantSentence>() {
            @Override
            public int compare(RelevantSentence s1, RelevantSentence s2) {
                return Integer.compare(s2.getScore(), s1.getScore());
            }
        });
        
        // Rebuild text
        StringBuilder sorted = new StringBuilder();
        for (RelevantSentence rs : relevantSentences) {
            sorted.append(rs.getSentence()).append(". ");
        }
        
        // Remove last period
        if (sorted.length() > 0) {
            sorted.setLength(sorted.length() - 1);
        }
        
        textViewContent.setText(sorted.toString());
        currentText = sorted.toString();
        
        textViewStatus.setText("Sorted by relevance to keywords");
        Toast.makeText(this, "Content sorted by relevance", 
                      Toast.LENGTH_SHORT).show();
        
        // Re-apply highlight if needed
        if (isHighlighted && !searchKeywords.isEmpty()) {
            highlightKeywords();
        }
    }
    
    // Inner class for relevance sorting
    private class RelevantSentence {
        private String sentence;
        private int score;
        
        public RelevantSentence(String sentence, int score) {
            this.sentence = sentence;
            this.score = score;
        }
        
        public String getSentence() { return sentence; }
        public int getScore() { return score; }
    }
}

package com.example.databaseapp;

import android.content.ContentValues;
import android.content.Context;
import android.database.Cursor;
import android.database.sqlite.SQLiteDatabase;
import android.database.sqlite.SQLiteOpenHelper;
import android.util.Log;
import java.util.ArrayList;
import java.util.List;

public class DatabaseHelper extends SQLiteOpenHelper {
    
    // Database Information
    private static final String DATABASE_NAME = "MyApp.db";
    private static final int DATABASE_VERSION = 1;
    
    // Table Name
    private static final String TABLE_USERS = "users";
    private static final String TABLE_PRODUCTS = "products";
    
    // Table Columns - Users
    private static final String COLUMN_USER_ID = "id";
    private static final String COLUMN_USER_NAME = "name";
    private static final String COLUMN_USER_EMAIL = "email";
    private static final String COLUMN_USER_PHONE = "phone";
    private static final String COLUMN_USER_AGE = "age";
    
    // Table Columns - Products
    private static final String COLUMN_PRODUCT_ID = "id";
    private static final String COLUMN_PRODUCT_NAME = "name";
    private static final String COLUMN_PRODUCT_PRICE = "price";
    private static final String COLUMN_PRODUCT_QUANTITY = "quantity";
    
    // Create Table Queries
    private static final String CREATE_TABLE_USERS = 
        "CREATE TABLE " + TABLE_USERS + "("
            + COLUMN_USER_ID + " INTEGER PRIMARY KEY AUTOINCREMENT,"
            + COLUMN_USER_NAME + " TEXT NOT NULL,"
            + COLUMN_USER_EMAIL + " TEXT UNIQUE,"
            + COLUMN_USER_PHONE + " TEXT,"
            + COLUMN_USER_AGE + " INTEGER"
        + ")";
    
    private static final String CREATE_TABLE_PRODUCTS = 
        "CREATE TABLE " + TABLE_PRODUCTS + "("
            + COLUMN_PRODUCT_ID + " INTEGER PRIMARY KEY AUTOINCREMENT,"
            + COLUMN_PRODUCT_NAME + " TEXT NOT NULL,"
            + COLUMN_PRODUCT_PRICE + " REAL,"
            + COLUMN_PRODUCT_QUANTITY + " INTEGER"
        + ")";
    
    // Constructor
    public DatabaseHelper(Context context) {
        super(context, DATABASE_NAME, null, DATABASE_VERSION);
    }
    
    @Override
    public void onCreate(SQLiteDatabase db) {
        // Create tables
        db.execSQL(CREATE_TABLE_USERS);
        db.execSQL(CREATE_TABLE_PRODUCTS);
        
        Log.d("DatabaseHelper", "Tables created successfully");
    }
    
    @Override
    public void onUpgrade(SQLiteDatabase db, int oldVersion, int newVersion) {
        // Drop older tables if exist
        db.execSQL("DROP TABLE IF EXISTS " + TABLE_USERS);
        db.execSQL("DROP TABLE IF EXISTS " + TABLE_PRODUCTS);
        
        // Create tables again
        onCreate(db);
    }
    
    // ============== CRUD OPERATIONS ==============
    
    // -------------------- USER OPERATIONS --------------------
    
    // Insert User
    public long insertUser(String name, String email, String phone, int age) {
        SQLiteDatabase db = this.getWritableDatabase();
        ContentValues values = new ContentValues();
        
        values.put(COLUMN_USER_NAME, name);
        values.put(COLUMN_USER_EMAIL, email);
        values.put(COLUMN_USER_PHONE, phone);
        values.put(COLUMN_USER_AGE, age);
        
        // Insert row
        long id = db.insert(TABLE_USERS, null, values);
        db.close();
        
        return id;
    }
    
    // Get Single User
    public User getUser(int id) {
        SQLiteDatabase db = this.getReadableDatabase();
        
        Cursor cursor = db.query(TABLE_USERS, 
            new String[]{COLUMN_USER_ID, COLUMN_USER_NAME, COLUMN_USER_EMAIL, 
                        COLUMN_USER_PHONE, COLUMN_USER_AGE},
            COLUMN_USER_ID + "=?",
            new String[]{String.valueOf(id)}, 
            null, null, null, null);
        
        User user = null;
        if (cursor != null && cursor.moveToFirst()) {
            user = new User(
                cursor.getInt(cursor.getColumnIndexOrThrow(COLUMN_USER_ID)),
                cursor.getString(cursor.getColumnIndexOrThrow(COLUMN_USER_NAME)),
                cursor.getString(cursor.getColumnIndexOrThrow(COLUMN_USER_EMAIL)),
                cursor.getString(cursor.getColumnIndexOrThrow(COLUMN_USER_PHONE)),
                cursor.getInt(cursor.getColumnIndexOrThrow(COLUMN_USER_AGE))
            );
            cursor.close();
        }
        db.close();
        
        return user;
    }
    
    // Get All Users
    public List<User> getAllUsers() {
        List<User> userList = new ArrayList<>();
        
        String selectQuery = "SELECT * FROM " + TABLE_USERS + " ORDER BY " + 
                            COLUMN_USER_NAME + " ASC";
        
        SQLiteDatabase db = this.getReadableDatabase();
        Cursor cursor = db.rawQuery(selectQuery, null);
        
        if (cursor.moveToFirst()) {
            do {
                User user = new User();
                user.setId(cursor.getInt(cursor.getColumnIndexOrThrow(COLUMN_USER_ID)));
                user.setName(cursor.getString(cursor.getColumnIndexOrThrow(COLUMN_USER_NAME)));
                user.setEmail(cursor.getString(cursor.getColumnIndexOrThrow(COLUMN_USER_EMAIL)));
                user.setPhone(cursor.getString(cursor.getColumnIndexOrThrow(COLUMN_USER_PHONE)));
                user.setAge(cursor.getInt(cursor.getColumnIndexOrThrow(COLUMN_USER_AGE)));
                
                userList.add(user);
            } while (cursor.moveToNext());
        }
        
        cursor.close();
        db.close();
        
        return userList;
    }
    
    // Update User
    public int updateUser(User user) {
        SQLiteDatabase db = this.getWritableDatabase();
        
        ContentValues values = new ContentValues();
        values.put(COLUMN_USER_NAME, user.getName());
        values.put(COLUMN_USER_EMAIL, user.getEmail());
        values.put(COLUMN_USER_PHONE, user.getPhone());
        values.put(COLUMN_USER_AGE, user.getAge());
        
        // Update row
        return db.update(TABLE_USERS, values, COLUMN_USER_ID + " = ?",
                new String[]{String.valueOf(user.getId())});
    }
    
    // Delete User
    public void deleteUser(User user) {
        SQLiteDatabase db = this.getWritableDatabase();
        db.delete(TABLE_USERS, COLUMN_USER_ID + " = ?",
                new String[]{String.valueOf(user.getId())});
        db.close();
    }
    
    // Get Users Count
    public int getUsersCount() {
        String countQuery = "SELECT * FROM " + TABLE_USERS;
        SQLiteDatabase db = this.getReadableDatabase();
        Cursor cursor = db.rawQuery(countQuery, null);
        
        int count = cursor.getCount();
        cursor.close();
        db.close();
        
        return count;
    }
    
    // -------------------- PRODUCT OPERATIONS --------------------
    
    // Insert Product
    public long insertProduct(String name, double price, int quantity) {
        SQLiteDatabase db = this.getWritableDatabase();
        ContentValues values = new ContentValues();
        
        values.put(COLUMN_PRODUCT_NAME, name);
        values.put(COLUMN_PRODUCT_PRICE, price);
        values.put(COLUMN_PRODUCT_QUANTITY, quantity);
        
        long id = db.insert(TABLE_PRODUCTS, null, values);
        db.close();
        
        return id;
    }
    
    // Get All Products
    public List<Product> getAllProducts() {
        List<Product> productList = new ArrayList<>();
        
        String selectQuery = "SELECT * FROM " + TABLE_PRODUCTS;
        
        SQLiteDatabase db = this.getReadableDatabase();
        Cursor cursor = db.rawQuery(selectQuery, null);
        
        if (cursor.moveToFirst()) {
            do {
                Product product = new Product();
                product.setId(cursor.getInt(cursor.getColumnIndexOrThrow(COLUMN_PRODUCT_ID)));
                product.setName(cursor.getString(cursor.getColumnIndexOrThrow(COLUMN_PRODUCT_NAME)));
                product.setPrice(cursor.getDouble(cursor.getColumnIndexOrThrow(COLUMN_PRODUCT_PRICE)));
                product.setQuantity(cursor.getInt(cursor.getColumnIndexOrThrow(COLUMN_PRODUCT_QUANTITY)));
                
                productList.add(product);
            } while (cursor.moveToNext());
        }
        
        cursor.close();
        db.close();
        
        return productList;
    }
    
    // Calculate Total Cost of All Products
    public double getTotalCost() {
        String query = "SELECT SUM(" + COLUMN_PRODUCT_PRICE + " * " + 
                      COLUMN_PRODUCT_QUANTITY + ") as total FROM " + TABLE_PRODUCTS;
        
        SQLiteDatabase db = this.getReadableDatabase();
        Cursor cursor = db.rawQuery(query, null);
        
        double total = 0;
        if (cursor.moveToFirst()) {
            total = cursor.getDouble(cursor.getColumnIndexOrThrow("total"));
        }
        
        cursor.close();
        db.close();
        
        return total;
    }
    
    // Delete Product
    public void deleteProduct(int id) {
        SQLiteDatabase db = this.getWritableDatabase();
        db.delete(TABLE_PRODUCTS, COLUMN_PRODUCT_ID + " = ?",
                new String[]{String.valueOf(id)});
        db.close();
    }
    
    // Clear all products
    public void deleteAllProducts() {
        SQLiteDatabase db = this.getWritableDatabase();
        db.execSQL("DELETE FROM " + TABLE_PRODUCTS);
        db.close();
    }
}

java
package com.example.databaseapp;

import android.os.Bundle;
import android.view.View;
import android.widget.AdapterView;
import android.widget.ArrayAdapter;
import android.widget.Button;
import android.widget.EditText;
import android.widget.ListView;
import android.widget.Spinner;
import android.widget.TextView;
import android.widget.Toast;
import androidx.appcompat.app.AppCompatActivity;
import java.text.DecimalFormat;
import java.util.ArrayList;
import java.util.List;

public class GroceryActivity extends AppCompatActivity {
    
    private Spinner spinnerItems;
    private EditText editTextQuantity;
    private Button btnAddItem, btnClear;
    private ListView listViewCart;
    private TextView textViewTotal;
    
    private DatabaseHelper dbHelper;
    private List<Product> cartItems;
    private ArrayAdapter<Product> cartAdapter;
    private DecimalFormat df = new DecimalFormat("#.##");
    
    // Predefined grocery items
    private String[] itemNames = {"Select Item", "Apple", "Banana", "Bread", "Milk", 
                                  "Eggs", "Rice", "Chicken", "Fish", "Potato", "Onion"};
    private double[] itemPrices = {0, 1.99, 0.99, 2.49, 3.49, 4.99, 10.99, 8.99, 12.99, 1.49, 1.29};
    
    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_grocery);
        
        initViews();
        setupSpinner();
        setupListeners();
        loadCartItems();
    }
    
    private void initViews() {
        spinnerItems = findViewById(R.id.spinnerItems);
        editTextQuantity = findViewById(R.id.editTextQuantity);
        btnAddItem = findViewById(R.id.btnAddItem);
        btnClear = findViewById(R.id.btnClear);
        listViewCart = findViewById(R.id.listViewCart);
        textViewTotal = findViewById(R.id.textViewTotal);
        
        dbHelper = new DatabaseHelper(this);
        cartItems = new ArrayList<>();
        
        // Clear any existing products (for demo)
        dbHelper.deleteAllProducts();
    }
    
    private void setupSpinner() {
        // Create adapter for spinner
        ArrayAdapter<String> spinnerAdapter = new ArrayAdapter<>(
            this,
            android.R.layout.simple_spinner_item,
            itemNames
        );
        spinnerAdapter.setDropDownViewResource(android.R.layout.simple_spinner_dropdown_item);
        spinnerItems.setAdapter(spinnerAdapter);
    }
    
    private void setupListeners() {
        btnAddItem.setOnClickListener(new View.OnClickListener() {
            @Override
            public void onClick(View v) {
                addItemToCart();
            }
        });
        
        btnClear.setOnClickListener(new View.OnClickListener() {
            @Override
            public void onClick(View v) {
                clearCart();
            }
        });
        
        // Long press to remove item
        listViewCart.setOnItemLongClickListener(new AdapterView.OnItemLongClickListener() {
            @Override
            public boolean onItemLongClick(AdapterView<?> parent, View view, 
                                           int position, long id) {
                removeItem(position);
                return true;
            }
        });
    }
    
    private void addItemToCart() {
        int selectedPosition = spinnerItems.getSelectedItemPosition();
        
        if (selectedPosition == 0) {
            Toast.makeText(this, "Please select an item", Toast.LENGTH_SHORT).show();
            return;
        }
        
        String quantityStr = editTextQuantity.getText().toString().trim();
        if (quantityStr.isEmpty()) {
            editTextQuantity.setError("Please enter quantity");
            return;
        }
        
        int quantity = Integer.parseInt(quantityStr);
        if (quantity <= 0) {
            Toast.makeText(this, "Quantity must be greater than 0", 
                          Toast.LENGTH_SHORT).show();
            return;
        }
        
        // Get selected item details
        String itemName = itemNames[selectedPosition];
        double price = itemPrices[selectedPosition];
        
        // Insert into database
        long id = dbHelper.insertProduct(itemName, price, quantity);
        
        if (id != -1) {
            Toast.makeText(this, "Item added to cart", Toast.LENGTH_SHORT).show();
            
            // Clear quantity
            editTextQuantity.setText("");
            spinnerItems.setSelection(0);
            
            // Refresh cart
            loadCartItems();
        } else {
            Toast.makeText(this, "Error adding item", Toast.LENGTH_SHORT).show();
        }
    }
    
    private void loadCartItems() {
        cartItems.clear();
        cartItems.addAll(dbHelper.getAllProducts());
        
        if (cartAdapter == null) {
            cartAdapter = new ArrayAdapter<>(this, 
                android.R.layout.simple_list_item_1, cartItems);
            listViewCart.setAdapter(cartAdapter);
        } else {
            cartAdapter.notifyDataSetChanged();
        }
        
        // Update total
        updateTotal();
    }
    
    private void removeItem(int position) {
        Product product = cartItems.get(position);
        dbHelper.deleteProduct(product.getId());
        
        Toast.makeText(this, "Removed: " + product.getName(), 
                      Toast.LENGTH_SHORT).show();
        
        loadCartItems();
    }
    
    private void clearCart() {
        if (cartItems.isEmpty()) {
            Toast.makeText(this, "Cart is already empty", Toast.LENGTH_SHORT).show();
            return;
        }
        
        dbHelper.deleteAllProducts();
        loadCartItems();
        
        Toast.makeText(this, "Cart cleared", Toast.LENGTH_SHORT).show();
    }
    
    private void updateTotal() {
        double total = dbHelper.getTotalCost();
        textViewTotal.setText("$" + df.format(total));
    }
}

    public MovieReview(String movieName, int year, float rating, String reviewText) {
        this.movieName = movieName;
        this.year = year;
        this.rating = rating;
        this.reviewText = reviewText;
    }

package com.example.moviereview;

import android.content.ContentValues;
import android.content.Context;
import android.database.Cursor;
import android.database.sqlite.SQLiteDatabase;
import android.database.sqlite.SQLiteOpenHelper;

public class MovieDatabaseHelper extends SQLiteOpenHelper {
    
    private static final String DATABASE_NAME = "MovieReviews.db";
    private static final int DATABASE_VERSION = 1;
    
    private static final String TABLE_MOVIES = "movies";
    private static final String COLUMN_ID = "id";
    private static final String COLUMN_NAME = "movie_name";
    private static final String COLUMN_YEAR = "year";
    private static final String COLUMN_RATING = "rating";
    private static final String COLUMN_REVIEW = "review";
    
    private static final String CREATE_TABLE = 
        "CREATE TABLE " + TABLE_MOVIES + "("
            + COLUMN_ID + " INTEGER PRIMARY KEY AUTOINCREMENT,"
            + COLUMN_NAME + " TEXT NOT NULL,"
            + COLUMN_YEAR + " INTEGER,"
            + COLUMN_RATING + " REAL,"
            + COLUMN_REVIEW + " TEXT"
        + ")";
    
    public MovieDatabaseHelper(Context context) {
        super(context, DATABASE_NAME, null, DATABASE_VERSION);
    }
    
    @Override
    public void onCreate(SQLiteDatabase db) {
        db.execSQL(CREATE_TABLE);
        
        // Insert sample data
        insertSampleMovies(db);
    }
    
    private void insertSampleMovies(SQLiteDatabase db) {
        String[][] sampleMovies = {
            {"The Shawshank Redemption", "1994", "5", "Amazing story of hope and friendship"},
            {"The Godfather", "1972", "5", "Masterpiece of cinema"},
            {"Pulp Fiction", "1994", "4.5", "Quentin Tarantino's best work"},
            {"The Dark Knight", "2008", "5", "Heath Ledger's performance is legendary"},
            {"Forrest Gump", "1994", "4.5", "Heartwarming and inspirational"},
            {"Inception", "2010", "4.5", "Mind-bending sci-fi thriller"},
            {"The Matrix", "1999", "4.5", "Revolutionary sci-fi action"},
            {"Goodfellas", "1990", "4.5", "Scorsese's best gangster film"}
        };
        
        for (String[] movie : sampleMovies) {
            ContentValues values = new ContentValues();
            values.put(COLUMN_NAME, movie[0]);
            values.put(COLUMN_YEAR, Integer.parseInt(movie[1]));
            values.put(COLUMN_RATING, Float.parseFloat(movie[2]));
            values.put(COLUMN_REVIEW, movie[3]);
            db.insert(TABLE_MOVIES, null, values);
        }
    }
    
    @Override
    public void onUpgrade(SQLiteDatabase db, int oldVersion, int newVersion) {
        db.execSQL("DROP TABLE IF EXISTS " + TABLE_MOVIES);
        onCreate(db);
    }
    
    // Insert review
    public long insertReview(MovieReview review) {
        SQLiteDatabase db = this.getWritableDatabase();
        ContentValues values = new ContentValues();
        
        values.put(COLUMN_NAME, review.getMovieName());
        values.put(COLUMN_YEAR, review.getYear());
        values.put(COLUMN_RATING, review.getRating());
        values.put(COLUMN_REVIEW, review.getReviewText());
        
        long id = db.insert(TABLE_MOVIES, null, values);
        db.close();
        return id;
    }
    
    // Get all movies for ListView
    public List<MovieReview> getAllMovies() {
        List<MovieReview> movieList = new ArrayList<>();
        
        String selectQuery = "SELECT * FROM " + TABLE_MOVIES + 
                            " ORDER BY " + COLUMN_NAME + " ASC";
        
        SQLiteDatabase db = this.getReadableDatabase();
        Cursor cursor = db.rawQuery(selectQuery, null);
        
        if (cursor.moveToFirst()) {
            do {
                MovieReview movie = new MovieReview();
                movie.setId(cursor.getInt(cursor.getColumnIndexOrThrow(COLUMN_ID)));
                movie.setMovieName(cursor.getString(cursor.getColumnIndexOrThrow(COLUMN_NAME)));
                movie.setYear(cursor.getInt(cursor.getColumnIndexOrThrow(COLUMN_YEAR)));
                movie.setRating(cursor.getFloat(cursor.getColumnIndexOrThrow(COLUMN_RATING)));
                movie.setReviewText(cursor.getString(cursor.getColumnIndexOrThrow(COLUMN_REVIEW)));
                
                movieList.add(movie);
            } while (cursor.moveToNext());
        }
        
        cursor.close();
        db.close();
        
        return movieList;
    }
    
    // Get single movie by ID
    public MovieReview getMovie(int id) {
        SQLiteDatabase db = this.getReadableDatabase();
        
        Cursor cursor = db.query(TABLE_MOVIES,
            new String[]{COLUMN_ID, COLUMN_NAME, COLUMN_YEAR, COLUMN_RATING, COLUMN_REVIEW},
            COLUMN_ID + "=?",
            new String[]{String.valueOf(id)},
            null, null, null, null);
        
        MovieReview movie = null;
        if (cursor != null && cursor.moveToFirst()) {
            movie = new MovieReview();
            movie.setId(cursor.getInt(cursor.getColumnIndexOrThrow(COLUMN_ID)));
            movie.setMovieName(cursor.getString(cursor.getColumnIndexOrThrow(COLUMN_NAME)));
            movie.setYear(cursor.getInt(cursor.getColumnIndexOrThrow(COLUMN_YEAR)));
            movie.setRating(cursor.getFloat(cursor.getColumnIndexOrThrow(COLUMN_RATING)));
            movie.setReviewText(cursor.getString(cursor.getColumnIndexOrThrow(COLUMN_REVIEW)));
            cursor.close();
        }
        
        db.close();
        return movie;
    }

    private void initViews() {
        editTextMovieName = findViewById(R.id.editTextMovieName);
        editTextYear = findViewById(R.id.editTextYear);
        editTextReview = findViewById(R.id.editTextReview);
        seekBarRating = findViewById(R.id.seekBarRating);
        textViewRating = findViewById(R.id.textViewRating);
        btnSaveReview = findViewById(R.id.btnSaveReview);
        listViewMovies = findViewById(R.id.listViewMovies);
        
        dbHelper = new MovieDatabaseHelper(this);
    }
    
    private void setupSeekBar() {
        seekBarRating.setOnSeekBarChangeListener(new SeekBar.OnSeekBarChangeListener() {
            @Override
            public void onProgressChanged(SeekBar seekBar, int progress, boolean fromUser) {
                // Convert 0-40 to 0.5-5.0 (in steps of 0.5)
                float rating = (progress / 8.0f) + 0.5f;
                textViewRating.setText("Rating: " + String.format("%.1f", rating));
            }
            
            @Override
            public void onStartTrackingTouch(SeekBar seekBar) {}
            
            @Override
            public void onStopTrackingTouch(SeekBar seekBar) {}
        });
    }
    
    private void saveReview() {
        String movieName = editTextMovieName.getText().toString().trim();
        String yearStr = editTextYear.getText().toString().trim();
        String review = editTextReview.getText().toString().trim();
        
        // Validate inputs
        if (movieName.isEmpty()) {
            editTextMovieName.setError("Movie name required");
            return;
        }
        
        if (yearStr.isEmpty()) {
            editTextYear.setError("Year required");
            return;
        }
        
        int year = Integer.parseInt(yearStr);
        float rating = (seekBarRating.getProgress() / 8.0f) + 0.5f;
        
        // Create review object
        MovieReview movieReview = new MovieReview(movieName, year, rating, review);
        
        // Save to database
        long id = dbHelper.insertReview(movieReview);
        
        if (id != -1) {
            Toast.makeText(this, "Review saved successfully!", Toast.LENGTH_SHORT).show();
            
            // Clear inputs
            editTextMovieName.setText("");
            editTextYear.setText("");
            editTextReview.setText("");
            seekBarRating.setProgress(20);
            
            // Refresh list
            loadMovieList();
        } else {
            Toast.makeText(this, "Error saving review", Toast.LENGTH_SHORT).show();
        }
    }
    
    private void loadMovieList() {
        movieList = dbHelper.getAllMovies();
        
        if (adapter == null) {
            adapter = new ArrayAdapter<>(this,
                android.R.layout.simple_list_item_1, movieList);
            listViewMovies.setAdapter(adapter);
        } else {
            adapter.clear();
            adapter.addAll(movieList);
            adapter.notifyDataSetChanged();
        }
    }
    
    private void showMovieDetails(MovieReview movie) {
        // Create intent to start details activity
        android.content.Intent intent = new Intent(this, MovieDetailsActivity.class);
        intent.putExtra("movie_id", movie.getId());
        startActivity(intent);
    }

package com.example.preferencesdemo;

import android.content.SharedPreferences;
import android.os.Bundle;
import android.view.View;
import android.widget.Button;
import android.widget.EditText;
import android.widget.TextView;
import android.widget.Toast;
import androidx.appcompat.app.AppCompatActivity;

public class SharedPreferencesActivity extends AppCompatActivity {
    
    private EditText editTextName, editTextEmail, editTextAge;
    private Button btnSave, btnLoad, btnClear;
    private TextView textViewDisplay;
    
    // Shared Preferences file name
    private static final String PREF_NAME = "UserPrefs";
    
    // Keys for data
    private static final String KEY_NAME = "name";
    private static final String KEY_EMAIL = "email";
    private static final String KEY_AGE = "age";
    private static final String KEY_IS_LOGGED_IN = "isLoggedIn";
    
    private SharedPreferences sharedPreferences;
    
    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_preferences);
        
        initViews();
        
        // Initialize SharedPreferences
        sharedPreferences = getSharedPreferences(PREF_NAME, MODE_PRIVATE);
        
        // Check if user was logged in before
        checkLoginStatus();
        
        btnSave.setOnClickListener(new View.OnClickListener() {
            @Override
            public void onClick(View v) {
                saveData();
            }
        });
        
        btnLoad.setOnClickListener(new View.OnClickListener() {
            @Override
            public void onClick(View v) {
                loadData();
            }
        });
        
        btnClear.setOnClickListener(new View.OnClickListener() {
            @Override
            public void onClick(View v) {
                clearData();
            }
        });
    }
    
    private void initViews() {
        editTextName = findViewById(R.id.editTextName);
        editTextEmail = findViewById(R.id.editTextEmail);
        editTextAge = findViewById(R.id.editTextAge);
        btnSave = findViewById(R.id.btnSave);
        btnLoad = findViewById(R.id.btnLoad);
        btnClear = findViewById(R.id.btnClear);
        textViewDisplay = findViewById(R.id.textViewDisplay);
    }
    
    private void saveData() {
        String name = editTextName.getText().toString().trim();
        String email = editTextEmail.getText().toString().trim();
        String ageStr = editTextAge.getText().toString().trim();
        
        if (name.isEmpty() || email.isEmpty() || ageStr.isEmpty()) {
            Toast.makeText(this, "Please fill all fields", Toast.LENGTH_SHORT).show();
            return;
        }
        
        int age = Integer.parseInt(ageStr);
        
        // Get editor to modify shared preferences
        SharedPreferences.Editor editor = sharedPreferences.edit();
        
        // Store data
        editor.putString(KEY_NAME, name);
        editor.putString(KEY_EMAIL, email);
        editor.putInt(KEY_AGE, age);
        editor.putBoolean(KEY_IS_LOGGED_IN, true);
        
        // Apply changes (asynchronous)
        editor.apply();
        // Or use commit() for synchronous (returns boolean)
        // editor.commit();
        
        Toast.makeText(this, "Data saved successfully!", Toast.LENGTH_SHORT).show();
        
        // Display saved data
        displayData("Saved: " + name + ", " + email + ", " + age);
    }
    
    private void loadData() {
        // Retrieve data with default values
        String name = sharedPreferences.getString(KEY_NAME, "No name found");
        String email = sharedPreferences.getString(KEY_EMAIL, "No email found");
        int age = sharedPreferences.getInt(KEY_AGE, 0);
        boolean isLoggedIn = sharedPreferences.getBoolean(KEY_IS_LOGGED_IN, false);
        
        // Set to EditTexts
        editTextName.setText(name);
        editTextEmail.setText(email);
        editTextAge.setText(String.valueOf(age));
        
        // Display
        String display = "Loaded Data:\n" +
                        "Name: " + name + "\n" +
                        "Email: " + email + "\n" +
                        "Age: " + age + "\n" +
                        "Logged In: " + (isLoggedIn ? "Yes" : "No");
        
        displayData(display);
        
        Toast.makeText(this, "Data loaded", Toast.LENGTH_SHORT).show();
    }
    
    private void clearData() {
        // Clear all data
        SharedPreferences.Editor editor = sharedPreferences.edit();
        editor.clear();
        editor.apply();
        
        // Clear EditTexts
        editTextName.setText("");
        editTextEmail.setText("");
        editTextAge.setText("");
        
        textViewDisplay.setText("All data cleared");
        
        Toast.makeText(this, "Data cleared", Toast.LENGTH_SHORT).show();
    }
    
    private void checkLoginStatus() {
        boolean isLoggedIn = sharedPreferences.getBoolean(KEY_IS_LOGGED_IN, false);
        
        if (isLoggedIn) {
            // User was logged in, load their data
            loadData();
            textViewDisplay.setText("Welcome back! Data loaded automatically.");
        }
    }
    
    private void displayData(String data) {
        textViewDisplay.setText(data);
    }
    
    // Example of removing specific key
    private void removeKey(String key) {
        SharedPreferences.Editor editor = sharedPreferences.edit();
        editor.remove(key);
        editor.apply();
    }

package com.example.preferencesdemo;

import android.content.SharedPreferences;
import android.os.Bundle;
import android.widget.CheckBox;
import android.widget.CompoundButton;
import android.widget.RadioButton;
import android.widget.RadioGroup;
import android.widget.SeekBar;
import android.widget.Switch;
import android.widget.TextView;
import androidx.appcompat.app.AppCompatActivity;

public class SettingsActivity extends AppCompatActivity {
    
    private static final String PREF_SETTINGS = "AppSettings";
    private static final String KEY_NOTIFICATIONS = "notifications";
    private static final String KEY_DARK_MODE = "darkMode";
    private static final String KEY_THEME_COLOR = "themeColor";
    private static final String KEY_FONT_SIZE = "fontSize";
    
    private Switch switchNotifications;
    private CheckBox checkBoxDarkMode;
    private RadioGroup radioGroupTheme;
    private RadioButton radioBlue, radioGreen, radioRed;
    private SeekBar seekBarFontSize;
    private TextView textViewFontSize;
    
    private SharedPreferences settingsPrefs;
    
    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_settings);
        
        initViews();
        
        settingsPrefs = getSharedPreferences(PREF_SETTINGS, MODE_PRIVATE);
        
        loadSettings();
        setupListeners();
    }
    
    private void initViews() {
        switchNotifications = findViewById(R.id.switchNotifications);
        checkBoxDarkMode = findViewById(R.id.checkBoxDarkMode);
        radioGroupTheme = findViewById(R.id.radioGroupTheme);
        radioBlue = findViewById(R.id.radioBlue);
        radioGreen = findViewById(R.id.radioGreen);
        radioRed = findViewById(R.id.radioRed);
        seekBarFontSize = findViewById(R.id.seekBarFontSize);
        textViewFontSize = findViewById(R.id.textViewFontSize);
    }
    
    private void loadSettings() {
        // Load saved values or defaults
        boolean notifications = settingsPrefs.getBoolean(KEY_NOTIFICATIONS, true);
        boolean darkMode = settingsPrefs.getBoolean(KEY_DARK_MODE, false);
        String themeColor = settingsPrefs.getString(KEY_THEME_COLOR, "blue");
        int fontSize = settingsPrefs.getInt(KEY_FONT_SIZE, 16);
        
        // Set to views
        switchNotifications.setChecked(notifications);
        checkBoxDarkMode.setChecked(darkMode);
        
        switch (themeColor) {
            case "blue":
                radioBlue.setChecked(true);
                break;
            case "green":
                radioGreen.setChecked(true);
                break;
            case "red":
                radioRed.setChecked(true);
                break;
        }
        
        seekBarFontSize.setProgress(fontSize - 10); // Map 10-30 to 0-20
        textViewFontSize.setText("Font Size: " + fontSize + "sp");
    }
    
    private void setupListeners() {
        // Switch listener
        switchNotifications.setOnCheckedChangeListener(new CompoundButton.OnCheckedChangeListener() {
            @Override
            public void onCheckedChanged(CompoundButton buttonView, boolean isChecked) {
                settingsPrefs.edit().putBoolean(KEY_NOTIFICATIONS, isChecked).apply();
            }
        });
        
        // CheckBox listener
        checkBoxDarkMode.setOnCheckedChangeListener(new CompoundButton.OnCheckedChangeListener() {
            @Override
            public void onCheckedChanged(CompoundButton buttonView, boolean isChecked) {
                settingsPrefs.edit().putBoolean(KEY_DARK_MODE, isChecked).apply();
            }
        });
        
        // RadioGroup listener
        radioGroupTheme.setOnCheckedChangeListener(new RadioGroup.OnCheckedChangeListener() {
            @Override
            public void onCheckedChanged(RadioGroup group, int checkedId) {
                String theme = "blue";
                if (checkedId == R.id.radioGreen) {
                    theme = "green";
                } else if (checkedId == R.id.radioRed) {
                    theme = "red";
                }
                settingsPrefs.edit().putString(KEY_THEME_COLOR, theme).apply();
            }
        });
        
        // SeekBar listener
        seekBarFontSize.setOnSeekBarChangeListener(new SeekBar.OnSeekBarChangeListener() {
            @Override
            public void onProgressChanged(SeekBar seekBar, int progress, boolean fromUser) {
                int fontSize = progress + 10; // Map 0-20 to 10-30
                textViewFontSize.setText("Font Size: " + fontSize + "sp");
                settingsPrefs.edit().putInt(KEY_FONT_SIZE, fontSize).apply();
            }
            
            @Override
            public void onStartTrackingTouch(SeekBar seekBar) {}
            
            @Override
            public void onStopTrackingTouch(SeekBar seekBar) {}
        });
    }
}

// Method to log database contents to Logcat
private void logDatabaseContents() {
    SQLiteDatabase db = dbHelper.getReadableDatabase();
    
    // Get all tables
    Cursor cursor = db.rawQuery("SELECT name FROM sqlite_master WHERE type='table'", null);
    
    if (cursor.moveToFirst()) {
        do {
            String tableName = cursor.getString(0);
            Log.d("Database", "Table: " + tableName);
            
            // Get table info
            Cursor tableCursor = db.rawQuery("PRAGMA table_info(" + tableName + ")", null);
            StringBuilder columns = new StringBuilder("Columns: ");
            while (tableCursor.moveToNext()) {
                columns.append(tableCursor.getString(1)).append(", ");
            }
            Log.d("Database", columns.toString());
            tableCursor.close();
            
            // Get row count
            Cursor countCursor = db.rawQuery("SELECT COUNT(*) FROM " + tableName, null);
            if (countCursor.moveToFirst()) {
                Log.d("Database", "Row count: " + countCursor.getInt(0));
            }
            countCursor.close();
            
        } while (cursor.moveToNext());
    }
    cursor.close();
    db.close();
}
